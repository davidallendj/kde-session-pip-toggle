cmake_minimum_required(VERSION 3.24)
project(kde-pip-session LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)
find_package(KF6GlobalAccel REQUIRED)
qt_add_executable(kde-pip-session src/main.cpp)
qt_add_executable(kde-pip-session-decoration src/kde-pip-session-decoration.cpp)
target_link_libraries(kde-pip-session-decoration PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
target_link_libraries(kde-pip-session PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets KF6::GlobalAccel)
install(TARGETS kde-pip-session RUNTIME DESTINATION bin)
install(TARGETS kde-pip-session-decoration RUNTIME DESTINATION libexec/kde-pip-session)
install(PROGRAMS scripts/nested-plasma-session DESTINATION libexec/kde-pip-session)
install(FILES kde-pip-session.desktop DESTINATION share/applications)
