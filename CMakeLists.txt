cmake_minimum_required(VERSION 3.24)
project(kde-toggle LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets DBus)
find_package(KF6GlobalAccel REQUIRED)
qt_add_executable(kde-toggle src/main.cpp)
qt_add_executable(nested-decoration src/nested-decoration.cpp)
target_link_libraries(nested-decoration PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
target_link_libraries(kde-toggle PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::DBus KF6::GlobalAccel)
install(TARGETS kde-toggle RUNTIME DESTINATION bin)
install(TARGETS nested-decoration RUNTIME DESTINATION libexec/kde-toggle)
install(PROGRAMS scripts/nested-plasma-session DESTINATION libexec/kde-toggle)
install(FILES kde-toggle.desktop DESTINATION share/applications)
